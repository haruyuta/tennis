<script>
"use strict";
// ================= 設定 =================
const GID = "14un6e";
const BASE = "http://chotto-ii.com/apps/mobile/?gid=" + GID + "&eid=";
const EVENTS_URL = "events.json"; // GitHub Actions が自動更新するデータファイル
// events.json が取得できない場合(オフライン等)の非常用データ。
// 通常は編集不要。events.json が正となる。
const FALLBACK = {
updated: "",
events: [
{date:"2026-07-13", eid:"344413"},
{date:"2026-07-15", eid:"344414"},
{date:"2026-07-22", eid:"344415"},
{date:"2026-07-27", eid:"344416"},
{date:"2026-07-29", eid:"344417"}
]
};
const REDIRECT_DELAY = 800; // 自動遷移までの表示時間(ms)
const LOW_THRESHOLD = 2; // 残り件数がこの値以下で警告
const STALE_DAYS = 3; // 自動更新が止まっているとみなす日数
// ========================================
const STORE_KEY = "nighter:user-events";
let userEvents = [];
try{
const s = localStorage.getItem(STORE_KEY);
if(s) userEvents = JSON.parse(s);
}catch(e){}
function saveUserEvents(){
try{ localStorage.setItem(STORE_KEY, JSON.stringify(userEvents)); }catch(e){}
}
const DOW = ["日","月","火","水","木","金","土"];
function fmt(dstr){
const d = new Date(dstr + "T00:00:00");
return {num:(d.getMonth()+1)+"/"+d.getDate(), dow:DOW[d.getDay()]};
}
function todayStr(){
const n = new Date();
return n.getFullYear()+"-"+String(n.getMonth()+1).padStart(2,"0")+"-"+String(n.getDate()).padStart(2,"0");
}
// ===== 読み込んだデータの状態 =====
let remoteEvents = []; // events.json 由来
let updatedAt = ""; // events.json の更新日時
let offline = false; // events.json の取得に失敗したか
const today = todayStr();
function allEvents(){
return [...remoteEvents, ...userEvents]
.filter((e,i,arr)=>arr.findIndex(x=>x.eid===e.eid)===i)
.sort((a,b)=>a.date.localeCompare(b.date));
}
async function loadEvents(){
try{
const res = await fetch(EVENTS_URL + "?t=" + Date.now(), {cache:"no-store"});
if(!res.ok) throw new Error("HTTP " + res.status);
const data = await res.json();
remoteEvents = data.events || [];
updatedAt = data.updated || "";
}catch(e){
offline = true;
remoteEvents = FALLBACK.events;
updatedAt = FALLBACK.updated;
}
}
function isStale(){
if(offline || !updatedAt) return offline;
const t = new Date(updatedAt).getTime();
if(isNaN(t)) return false;
return (Date.now() - t) > STALE_DAYS * 86400000;
}
async function main(){
await loadEvents();
const params = new URLSearchParams(location.search);
const menuMode = params.has("menu") || location.hash === "#menu";
const upcoming = allEvents().filter(e=>e.date >= today);
if(!menuMode && upcoming.length){
const e = upcoming[0];
const f = fmt(e.date);
const isToday = e.date === today;
document.getElementById("app").innerHTML =
'<div class="splash">' +
'<div style="font-size:12px;letter-spacing:0.2em;color:var(--lamp);">月水ナイター</div>' +
'<div class="date">' + f.num + '</div>' +
'<div class="dow">(' + f.dow + ') ' + (isToday ? '今日の練習' : '次回の練習') + '</div>' +
'<div class="note">出欠ページへ移動中…<br>' +
'<a href="' + BASE + e.eid + '">移動しない場合はこちらをタップ</a><br><br>' +
'<a href="?menu=1" style="color:var(--muted);">一覧・設定を開く(自動移動を止める)</a></div>' +
'</div>';
setTimeout(()=>{ location.replace(BASE + e.eid); }, REDIRECT_DELAY);
}else{
renderMenu(upcoming);
}
}
function renderMenu(upcoming){
const past = allEvents().filter(e=>e.date < today).reverse();
const all = allEvents();
const lastDate = all.length ? all[all.length-1].date : null;
const updDisp = updatedAt ? updatedAt.replace("T"," ").slice(0,16) : "不明";
let html =
'<header><div class="eyebrow">月水ナイター</div><h1>出欠確認ランチャー</h1>' +
'<div class="sub">今日: ' + fmt(today).num + ' (' + fmt(today).dow + ')</div>' +
(lastDate ? '<div class="lastdate">イベント: ' + fmt(lastDate).num + ' (' + fmt(lastDate).dow + ') まで / 自動更新: ' + updDisp + '</div>' : '') +
'</header>';
const boardTop = 'http://chotto-ii.com/apps/mobile/?gid=' + GID;
if(offline){
html += '<div class="warn offline">イベントデータ(events.json)を取得できませんでした。非常用データで表示しています。通信環境を確認してください。</div>';
}else if(isStale()){
html += '<div class="warn low">自動更新が ' + STALE_DAYS + ' 日以上行われていません。GitHub Actions の実行状況を確認してください。</div>';
}
if(upcoming.length === 0){
html += '<div class="warn empty">今後の練習が登録されていません。' +
'<a href="' + boardTop + '">掲示板</a>で最新のイベントを確認してください(掲示板に登録されれば翌朝に自動反映されます)。急ぎの場合は下のフォームから追加できます。</div>';
}else if(upcoming.length <= LOW_THRESHOLD){
html += '<div class="warn low">今後の練習が残り ' + upcoming.length + ' 件です。掲示板に次回イベントが登録されれば自動で反映されます。</div>';
}
if(upcoming.length){
html += '<div class="section-label">今後の練習</div>';
upcoming.forEach((e,idx)=>{
const f = fmt(e.date);
let badge = "";
if(e.date === today) badge = '<span class="badge today">今日</span>';
else if(idx===0) badge = '<span class="badge next">次回</span>';
html += '<a class="event' + (idx===0?' hero':'') + '" href="' + BASE + e.eid + '">' +
f.num + '<span class="d-dow">(' + f.dow + ') 練習</span>' + badge + '</a>';
if(e.user){
html += '<button class="del" data-eid="' + e.eid + '">この追加イベントを削除</button>';
}
});
}else{
html += '<div class="section-label">今後のイベントが登録されていません。</div>';
}
if(past.length){
html += '<details><summary>終了した練習(' + past.length + '件)</summary>';
past.forEach(e=>{
const f = fmt(e.date);
html += '<a class="event past" href="' + BASE + e.eid + '">' +
f.num + '<span class="d-dow">(' + f.dow + ') 練習</span></a>';
});
html += '</details>';
}
html += '<a class="board-link" href="' + boardTop + '">掲示板トップを開く(全イベント一覧)</a>';
html +=
'<div class="addbox"><h2>イベントを追加(この端末のみ・応急用)</h2>' +
'<label>イベントページのURL</label>' +
'<input type="text" id="urlInput" placeholder="http://chotto-ii.com/apps/mobile/?gid=14un6e&eid=343500" inputmode="url">' +
'<label>練習日</label><input type="date" id="dateInput">' +
'<button id="addBtn" type="button">追加する</button>' +
'<div class="msg" id="addMsg"></div>' +
'<div style="font-size:11px;color:var(--muted);margin-top:10px;">通常は掲示板の内容が毎朝自動で反映されるため、このフォームは自動更新を待てない場合のみ使用してください。</div></div>';
document.getElementById("app").innerHTML = html;
document.getElementById("addBtn").onclick = ()=>{
const msg = document.getElementById("addMsg");
const url = document.getElementById("urlInput").value.trim();
const date = document.getElementById("dateInput").value;
const m = url.match(/eid=(\d+)/);
if(!m){ msg.textContent="URLからイベントID(eid=数字)が見つかりません。"; msg.className="msg err"; return; }
if(!date){ msg.textContent="練習日を選択してください。"; msg.className="msg err"; return; }
userEvents.push({date:date, eid:m[1], user:true});
saveUserEvents();
location.href = "?menu=1";
};
document.querySelectorAll(".del").forEach(btn=>{
btn.onclick = ()=>{
userEvents = userEvents.filter(x=>x.eid!==btn.dataset.eid);
saveUserEvents();
location.href = "?menu=1";
};
});
}
main();
</script>
